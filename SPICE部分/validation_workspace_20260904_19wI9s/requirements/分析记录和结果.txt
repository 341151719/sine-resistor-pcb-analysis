下面给你一份**可直接运行**的 Python 脚本。  
它采用的是 **Python 调用 ngspice 批处理** 的方式，而不是 PySpice 绑定方式。这样做的优点是：

1. **和 KiCad / ngspice 的工作流更一致**
2. **更稳**，少很多 PySpice 安装兼容性问题
3. 后面你要并入 **KiCad / Quilter** 流程时更方便

这份脚本会完成你前面文档里需要的核心仿真内容：

- **理想合成阻抗模型**
- **实现级行为模型**（电流检测 + 乘法器 + 功率跟随器）
- **固定目标阻值验证**：100 Ω、20 Ω、7.2 Ω、5 Ω、1 Ω
- **单正弦调制**
- **双正弦叠加调制**
- 自动输出：
  - `.cir` netlist
  - ngspice `.log`
  - `.dat` 原始数据
  - `.csv` 结果表
  - `.png` 图像
  - `.json` 摘要
  - 总汇总 `all_summary.csv`

@dataclass
class SpeakerTS:
    """用户给定 / 文档推导出的 T/S 参数"""
    Re: float = 7.2          # Ω
    Le: float = 0.2e-3       # H
    Bl: float = 4.6          # T·m
    Mms: float = 5.7e-3      # kg
    fs: float = 90.0         # Hz
    Qms: float = 2.29
    Sd: float = 50e-4        # 50 cm^2 = 0.005 m^2

    @property
    def w0(self) -> float:
        return 2.0 * math.pi * self.fs

    @property
    def Cms(self) -> float:
        return 1.0 / (self.Mms * self.w0 * self.w0)

    @property
    def Kms(self) -> float:
        return 1.0 / self.Cms

    @property
    def Rms(self) -> float:
        return self.w0 * self.Mms / self.Qms


@dataclass
class SimConfig:
    """
    仿真配置：
    Rtarget(t) = R0 + A1*sin(2π fm1 t + ph1) + A2*sin(2π fm2 t + ph2)
    再裁剪到 [RMIN, RMAX]
    """
    R0: float = 50.5
    A1: float = 49.5
    fm1: float = 100.0
    ph1: float = 0.0          # rad

    A2: float = 0.0
    fm2: float = 317.0
    ph2: float = 0.0          # rad

    RMIN: float = 1.0
    RMAX: float = 100.0

    Pamp: float = 1.0         # Pa，1 Pa ≈ 94 dB SPL
    fa: float = 120.0         # 外界声压频率 Hz

    tstop: float = 20e-3
    tstep: float = 2e-6

    # 实现级行为模型参数
    RSH: float = 0.1          # 电流检测分流电阻
    VDRVLIM: float = 16.0     # 功率跟随器等效输出饱和
    RSNUB: float = 4.7
    CSNUB: float = 47e-9

* -----------------------------

def sf(x: float) -> str:
    """数字转为 SPICE 友好的字符串"""
    return f"{float(x):.12g}"


def safe_name(x: float) -> str:
    s = f"{x:g}"
    return s.replace("-", "m").replace(".", "p")


def fmt_val(x) -> str:
    try:
        xf = float(x)
        if math.isnan(xf) or math.isinf(xf):
            return "nan"
        return f"{xf:.4g}"
    except Exception:
        return str(x)


def render_template(template: str, mapping: dict) -> str:
    txt = textwrap.dedent(template).strip() + "\n"
    for k, v in mapping.items():
        txt = txt.replace(f"__{k}__", str(v))
    return txt


def finite_only(a):
    arr = np.asarray(a, dtype=float)
    return arr[np.isfinite(arr)]


def nanmin_f(a) -> float:
    b = finite_only(a)
    return float(np.min(b)) if b.size else float("nan")


def nanmax_f(a) -> float:
    b = finite_only(a)
    return float(np.max(b)) if b.size else float("nan")


def nanmean_f(a) -> float:
    b = finite_only(a)
    return float(np.mean(b)) if b.size else float("nan")


def nanrms_f(a) -> float:
    b = finite_only(a)
    return float(np.sqrt(np.mean(b * b))) if b.size else float("nan")


def combine_legends(ax1, ax2=None, loc="best", ncol=1):
    h1, l1 = ax1.get_legend_handles_labels()
    if ax2 is None:
        ax1.legend(h1, l1, loc=loc, ncol=ncol)
    else:
        h2, l2 = ax2.get_legend_handles_labels()
        ax1.legend(h1 + h2, l1 + l2, loc=loc, ncol=ncol)


def check_ngspice():
    try:
        r = subprocess.run(
            [NGSPICE_BIN, "-v"],
            check=False,
            capture_output=True,
            text=True
        )
    except FileNotFoundError as e:
        raise FileNotFoundError(
            f"未找到 ngspice。请先安装 ngspice，或者设置环境变量 NGSPICE_BIN。\n"
            f"当前 NGSPICE_BIN = {NGSPICE_BIN}"
        ) from e

    if r.returncode != 0:
        raise RuntimeError(
            f"ngspice 可执行文件存在，但执行失败。\n"
            f"stdout:\n{r.stdout}\n\nstderr:\n{r.stderr}"
        )


def theory_for_target(spk: SpeakerTS, rtarget: float) -> tuple[float, float]:
    rem = (spk.Bl ** 2) / rtarget
    qeff = spk.w0 * spk.Mms / (spk.Rms + rem)
    return rem, qeff

def build_ideal_netlist(spk: SpeakerTS, cfg: SimConfig, datafile: str) -> str:
    return render_template(IDEAL_TEMPLATE, build_mapping(spk, cfg, datafile))


def build_impl_netlist(spk: SpeakerTS, cfg: SimConfig, datafile: str) -> str:
    return render_template(IMPLEMENTATION_TEMPLATE, build_mapping(spk, cfg, datafile))


def run_ngspice_case(netlist_text: str, workdir: Path, stem: str):
    workdir.mkdir(parents=True, exist_ok=True)
    cir_path = workdir / f"{stem}.cir"
    log_path = workdir / f"{stem}.log"

    cir_path.write_text(netlist_text, encoding="utf-8")

    result = subprocess.run(
        [NGSPICE_BIN, "-b", "-o", str(log_path), str(cir_path)],
        cwd=workdir,
        check=False,
        capture_output=True,
        text=True
    )

    if result.returncode != 0:
        log_txt = log_path.read_text(encoding="utf-8", errors="ignore") if log_path.exists() else ""
        raise RuntimeError(
            f"ngspice 运行失败: {stem}\n"
            f"returncode={result.returncode}\n"
            f"stdout:\n{result.stdout}\n\n"
            f"stderr:\n{result.stderr}\n\n"
            f"log:\n{log_txt}"
        )

    return cir_path, log_path


def read_wrdata(path: Path) -> pd.DataFrame:
    if not path.exists():
        raise FileNotFoundError(f"未找到 ngspice 输出数据文件: {path}")

    df = pd.read_csv(path, sep=r"\s+", engine="python")
    df.columns = [str(c).strip().lower() for c in df.columns]

    # 清理可能出现的重复/无用列
    drop_cols = []
    for c in df.columns:
        if c.startswith("unnamed"):
            drop_cols.append(c)
        if c == "index":
            drop_cols.append(c)
        if c.startswith("time.") or c.startswith("scale."):
            drop_cols.append(c)
    if drop_cols:
        df = df.drop(columns=drop_cols)

    if "time" not in df.columns:
        raise RuntimeError(
            f"wrdata 解析失败，未发现 time 列。\n"
            f"文件: {path}\n"
            f"当前列名: {df.columns.tolist()}"
        )

    return df


def add_derived_columns(df: pd.DataFrame, spk: SpeakerTS) -> pd.DataFrame:
    """
    从原始 wrdata 添加更稳健的派生量：
    - reff_calc: 用 Python 重新计算的等效总电阻（对 Iport≈0 做掩码）
    - reff_plot: 对绘图友好的插值版本
    - err_calc / err_plot
    - port_power_w
    - rem_theory / qeff_theory
    """
    out = df.copy()

    time = out["time"].to_numpy(dtype=float)
    rt = out["v(rtnode)"].to_numpy(dtype=float)
    vspk = out["v(spk)"].to_numpy(dtype=float)
    iport = out["v(iportnode)"].to_numpy(dtype=float)

    # 避免 I≈0 时比值发散
    imax = np.nanmax(np.abs(iport)) if len(iport) else 0.0
    ieps = max(1e-10, 1e-4 * float(imax))

    reff = np.full_like(vspk, np.nan, dtype=float)
    mask = np.abs(iport) > ieps
    with np.errstate(divide="ignore", invalid="ignore"):
        reff[mask] = spk.Re + vspk[mask] / iport[mask]

    out["reff_calc"] = reff
    out["reff_plot"] = pd.Series(reff).interpolate(limit_direction="both").to_numpy()
    out["err_calc"] = out["reff_calc"] - rt
    out["err_plot"] = out["reff_plot"] - rt
    out["port_power_w"] = vspk * iport
    out["rem_theory"] = (spk.Bl ** 2) / rt
    out["qeff_theory"] = spk.w0 * spk.Mms / (spk.Rms + out["rem_theory"])
    out["valid_reff_mask"] = mask.astype(int)
    out["iport_eps_used"] = ieps

    return out


def summary_from_df(df: pd.DataFrame, spk: SpeakerTS, cfg: SimConfig, case_name: str, model: str) -> dict:
    t = df["time"].to_numpy(dtype=float)
    tail_mask = t >= (0.5 * np.max(t))
    valid_mask = df["valid_reff_mask"].to_numpy(dtype=bool)
    valid_tail = tail_mask & valid_mask

    summary = {
        "case": case_name,
        "model": model,

        "rtarget_min_ohm": nanmin_f(df["v(rtnode)"]),
        "rtarget_max_ohm": nanmax_f(df["v(rtnode)"]),
        "rx_ext_min_ohm": nanmin_f(df["v(rxnode)"]),
        "rx_ext_max_ohm": nanmax_f(df["v(rxnode)"]),

        "reff_mean_tail_ohm": nanmean_f(df.loc[valid_tail, "reff_calc"]),
        "reff_min_valid_ohm": nanmin_f(df.loc[valid_mask, "reff_calc"]),
        "reff_max_valid_ohm": nanmax_f(df.loc[valid_mask, "reff_calc"]),

        "err_rms_tail_ohm": nanrms_f(df.loc[valid_tail, "err_calc"]),
        "err_abs_max_tail_ohm": nanmax_f(np.abs(df.loc[valid_tail, "err_calc"])),

        "iport_pk_a": nanmax_f(np.abs(df["v(iportnode)"])),
        "vspk_pk_v": nanmax_f(np.abs(df["v(spk)"])),
        "port_power_pk_w": nanmax_f(np.abs(df["port_power_w"])),
        "vel_rms_tail_mps": nanrms_f(df.loc[tail_mask, "v(vel)"]),
        "vel_pk_mps": nanmax_f(np.abs(df["v(vel)"])),

        "rem_min_ns_per_m": nanmin_f(df["v(remnode)"]),
        "rem_max_ns_per_m": nanmax_f(df["v(remnode)"]),
        "qeff_min": nanmin_f(df["v(qnode)"]),
        "qeff_max": nanmax_f(df["v(qnode)"]),

        "valid_reff_points": int(np.sum(valid_mask)),
        "total_points": int(len(df)),
        "iport_eps_used_a": float(df["iport_eps_used"].iloc[0]),
    }

    if abs(cfg.A1) < 1e-15 and abs(cfg.A2) < 1e-15:
        rem, qeff = theory_for_target(spk, cfg.R0)
        summary["fixed_target_ohm"] = float(cfg.R0)
        summary["theory_rem_ns_per_m"] = float(rem)
        summary["theory_qeff"] = float(qeff)

    return summary


def plot_case(df: pd.DataFrame, case_name: str, out_png: Path):
    t_ms = df["time"].to_numpy(dtype=float) * 1e3

    fig, axes = plt.subplots(4, 1, figsize=(13, 11), sharex=True)

    # 1) Rtarget / Reff / Rx
    ax = axes[0]
    ax.plot(t_ms, df["v(rtnode)"], label="Rtarget [Ω]", lw=1.8)
    ax.plot(t_ms, df["reff_plot"], label="Reff(calc) [Ω]", lw=1.4, ls="--")
    ax.plot(t_ms, df["v(rxnode)"], label="Rx(ext) [Ω]", lw=1.2, ls=":")
    ax.set_ylabel("Resistance")
    ax.set_title(case_name)
    combine_legends(ax, None, loc="best", ncol=3)

    # 2) Vspk / Iport / Vdrv / Vcmd
    ax = axes[1]
    ax.plot(t_ms, df["v(iportnode)"], label="Iport [A]", color="tab:blue", lw=1.2)
    ax.set_ylabel("Iport [A]")
    ax2 = ax.twinx()
    ax2.plot(t_ms, df["v(spk)"], label="Vspk [V]", color="tab:red", lw=1.2)
    if "v(drv)" in df.columns:

### 4. 力学趋势也是对的

这个趋势是：

这和理论完全一致。

因为电磁附加阻尼：

\[
R_\text{em}=\frac{(B\ell)^2}{R_\text{target}}
\]

随着 \(R_\text{target}\) 变小而变大。

按你的参数：

\[
(B\ell)^2 = 4.6^2 = 21.16
\]

所以：

- \(R_\text{target}=100\Omega\) 时，\(R_\text{em}\approx 0.212\)
- \(R_\text{target}=20\Omega\) 时，\(R_\text{em}\approx 1.058\)
- \(R_\text{target}=7.2\Omega\) 时，\(R_\text{em}\approx 2.94\)
- \(R_\text{target}=5\Omega\) 时，\(R_\text{em}\approx 4.23\)
- \(R_\text{target}=1\Omega\) 时，\(R_\text{em}\approx 21.16\)

因此速度被压低是完全正确的。

注意这里**不会严格按 100 倍**下降，因为系统总阻尼不只包含电磁阻尼，还包含：

- 机械损耗 \(R_\mathrm{ms}\)
- 非谐振点激励频率影响
- 线圈电感和系统动态项

所以你看到速度从 `0.002315` 降到 `0.0002404`，这是**很合理的**，不是异常。

---

### 5. 电流方向也合理

固定阻值时的峰值电流：

- `100 Ω` → `0.0001065 A`
- `20 Ω` → `0.0004011 A`
- `7.2 Ω` → `0.0007177 A`
- `5 Ω` → `0.0008371 A`
- `1 Ω` → `0.001094 A`

也是**阻抗越低，电流越大**。方向正确。

另外，`100 Ω` 这组误差相对稍大一点：

- `err_rms_tail = 0.0009266 Ω`

这其实不奇怪，因为在高阻状态下，`Iport` 很小，拿 `V/I` 重建电阻时会更容易受数值阈值和插值影响。  
但它仍然不到 1 mΩ，所以完全可以接受。
